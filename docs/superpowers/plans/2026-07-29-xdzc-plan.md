# 修仙搜打撤 Implementation Plan

> **For agentic workers:** This plan assumes zero context for the codebase. Each task is bite-sized (one action). Steps use checkbox (`- [ ]`) syntax.

**Goal:** Build a complete web-based idle cultivation treasure-hunting game (纯 HTML/CSS/JS, zero dependencies, double-click to run).

**Architecture:** Single HTML entry with modular JS files for each subsystem. CSS provides dark cultivation theme. localStorage for save/load. All data in config.js, all runtime state in state.js.

**Tech Stack:** HTML5, CSS3, Vanilla JavaScript, localStorage

      });
    });

- [ ] **Step 1.4: Create js/state.js with player state and save/load**

Create `js/state.js`:

```javascript
const STATE = {
  player: null,

  init() {
    const saved = localStorage.getItem("xdzc_save");
    if (saved) {
      try {
        const parsed = JSON.parse(saved);
        this.player = parsed.data || parsed;
        this.calculateOfflineReward();
      } catch (e) {
        this.createNewPlayer();
      }
    } else {
      this.createNewPlayer();
    }
    this.startAutoSave();
  },

  createNewPlayer() {
    this.player = {
      spiritStones: 100,
      currentRealm: 0,
      currentLayer: 1,
      cultivation: 0,
      rootBone: 1,
      comprehension: 1,
      luck: 1,
      storageBag: { items: [], capacity: CONFIG.BAG_CAPACITY },
      storageRing: { items: [], capacity: CONFIG.RING_CAPACITY },
      unlockedMaps: [1],
      currentMap: null,
      currentPath: [],
      pathIndex: 0,
      inExploration: false,
      lastOnline: Date.now()
    };
  },

  calculateOfflineReward() {
    const now = Date.now();
    const elapsed = Math.min(now - this.player.lastOnline, CONFIG.OFFLINE_MAX_HOURS * 3600000);
    const gain = this.player.comprehension * CONFIG.MEDITATION_BASE * (elapsed / 1000);
    this.player.cultivation += Math.floor(gain);
    this.player.lastOnline = now;
    return { hours: elapsed / 3600000, cultivation: Math.floor(gain) };
  },

  save() {
    this.player.lastOnline = Date.now();
    localStorage.setItem("xdzc_save", JSON.stringify({ version: 1, data: this.player }));
  },

  startAutoSave() {
    setInterval(() => this.save(), CONFIG.SAVE_INTERVAL_MS);
  },

  reset() {
    localStorage.removeItem("xdzc_save");
    this.createNewPlayer();
  }
};
```

const CULTIVATION = {
  getRealmName() {
    const p = STATE.player;
    return CONFIG.REALMS[p.currentRealm] + " " + p.currentLayer + "层";
  },

  getBreakthroughCost() {
    const p = STATE.player;
    return Math.floor(1000 * Math.pow(1.5, p.currentRealm * 9 + p.currentLayer - 1));
  },

  getMaxCultivation() {
    const p = STATE.player;
    return Math.floor(5000 * Math.pow(1.5, p.currentRealm * 9 + p.currentLayer - 1));
  },

  getMeditationRate() {
    return STATE.player.comprehension * CONFIG.MEDITATION_BASE;
  },

  tick() {
    const p = STATE.player;
    if (p.currentRealm < CONFIG.REALMS.length - 1 || p.currentLayer < 9) {
      p.cultivation += this.getMeditationRate() / 10;
    }
  },

  canBreakthrough() {
    const p = STATE.player;
    return p.cultivation >= this.getMaxCultivation();
  },

  breakthrough() {
    const p = STATE.player;
    if (!this.canBreakthrough()) return false;
    p.cultivation = 0;
    p.currentLayer++;
    if (p.currentLayer > 9) {
      p.currentLayer = 1;
      p.currentRealm++;
      const mapId = p.currentRealm + 1;
      if (!p.unlockedMaps.includes(mapId) && mapId <= CONFIG.MAPS.length) {
        p.unlockedMaps.push(mapId);
      }
    }
    return true;
  },

  upgradeAttribute(attr, cost) {
    const p = STATE.player;
    if (p.spiritStones < cost) return false;
    p.spiritStones -= cost;
    p[attr]++;
    return true;
  }
};
```

const EXPLORATION = {
  selectedMap: null,
  plannedPath: [],

  renderMapSelect() {
    const app = document.getElementById("app");
    app.dataset.view = "explore";
    let html = `
      <div class="panel">
        <h2 style="color:var(--accent);margin-bottom:12px;">选择探索地图</h2>
        <div style="display:grid;grid-template-columns:1fr 1fr;gap:8px;">
    `;
    CONFIG.MAPS.forEach(map => {
      const unlocked = STATE.player.unlockedMaps.includes(map.id);
      const affordable = STATE.player.spiritStones >= map.cost;
      html += `
        <div class="panel" style="opacity:${unlocked ? 1 : 0.5};cursor:${unlocked && affordable ? 'pointer' : 'not-allowed'};"
             onclick="${unlocked && affordable ? `EXPLORATION.selectMap(${map.id})` : ''}">
          <div style="font-weight:bold;color:var(--accent);">${map.name}</div>
          <div class="text-dim">${CONFIG.REALMS[map.realm]}</div>
          <div>${map.cost === 0 ? '免费' : map.cost.toLocaleString() + '灵石'}</div>
          <div class="text-dim" style="font-size:12px;">${map.valueMin.toLocaleString()} - ${map.valueMax.toLocaleString()}</div>
        </div>
      `;
    });
    html += `</div></div>`;
    html += `<button class="btn" style="width:100%;" onclick="MAIN.switchView('cave')">返回洞府</button>`;
    app.innerHTML = html;
  },

  selectMap(mapId) {
    const map = CONFIG.MAPS.find(m => m.id === mapId);
    if (!map || STATE.player.spiritStones < map.cost) return;
    this.selectedMap = map;
    this.plannedPath = [];
    this.renderPathPlanning();
  },

  renderPathPlanning() {
    const app = document.getElementById("app");
    const map = this.selectedMap;
    app.innerHTML = `
      <div class="panel">
        <h2 style="color:var(--accent);">${map.name} · 路径规划</h2>
        <div class="text-dim">选择3-5个点位进行探索</div>
      </div>
      <div class="panel" id="path-map" style="position:relative;height:300px;"></div>
      <div class="panel" id="path-info"></div>
      <div style="display:flex;gap:8px;">
        <button class="btn" style="flex:1;" onclick="EXPLORATION.autoPath()">自动路径</button>
        <button class="btn" style="flex:1;" onclick="EXPLORATION.startExploration()">开始探索</button>
        <button class="btn" style="flex:1;" onclick="EXPLORATION.renderMapSelect()">返回</button>
      </div>
    `;
    this.renderPointMap();
    this.updatePathInfo();
  },

  renderPointMap() {
    const container = document.getElementById("path-map");
    const points = [];
    for (let i = 0; i < 7; i++) {
      const angle = (i / 7) * Math.PI * 2 - Math.PI / 2;
      const x = 50 + 35 * Math.cos(angle);
      const y = 50 + 35 * Math.sin(angle);
      points.push({ id: i + 1, x, y });
    }
    let html = "";
    points.forEach((p, i) => {
      const selected = this.plannedPath.includes(p.id);
      const order = this.plannedPath.indexOf(p.id);
      html += `
        <div style="position:absolute;left:${p.x}%;top:${p.y}%;transform:translate(-50%,-50%);width:40px;height:40px;border-radius:50%;background:${selected ? 'var(--accent)' : 'var(--border)'};display:flex;align-items:center;justify-content:center;cursor:pointer;border:2px solid ${selected ? '#fff' : 'transparent'};"
             onclick="EXPLORATION.togglePoint(${p.id})">
          <span style="color:${selected ? '#1a1a1f' : 'var(--text)'};font-weight:bold;">${order >= 0 ? order + 1 : p.id}</span>
        </div>
      `;
    });
    container.innerHTML = html;
  },

  togglePoint(pointId) {
    const idx = this.plannedPath.indexOf(pointId);
    if (idx >= 0) {
      this.plannedPath.splice(idx, 1);
    } else if (this.plannedPath.length < 5) {
      this.plannedPath.push(pointId);
    }
    this.renderPointMap();
    this.updatePathInfo();
  },

  autoPath() {
    const count = 3 + Math.floor(Math.random() * 3);
    const available = [1, 2, 3, 4, 5, 6, 7];
    this.plannedPath = [];
    for (let i = 0; i < count; i++) {
      const idx = Math.floor(Math.random() * available.length);
      this.plannedPath.push(available.splice(idx, 1)[0]);
    }
    this.renderPointMap();
    this.updatePathInfo();
  },

  updatePathInfo() {
    const el = document.getElementById("path-info");
    if (!el) return;
    if (this.plannedPath.length === 0) {
      el.innerHTML = `<div class="text-dim">尚未选择点位</div>`;
      return;
    }
    const time = this.plannedPath.length * 10;
    el.innerHTML = `
      <div>已选路径：${this.plannedPath.join(" → ")}</div>
      <div class="text-dim">预估时间：${time}秒 &nbsp; 预估事件：${this.plannedPath.length - 1}~${this.plannedPath.length + 1}次</div>
    `;
  },

  startExploration() {
    if (this.plannedPath.length < 3) {
      alert("至少选择3个点位");
      return;
    }
    const map = this.selectedMap;
    STATE.player.spiritStones -= map.cost;
    STATE.player.currentMap = map.id;
    STATE.player.currentPath = [...this.plannedPath];
    STATE.player.pathIndex = 0;
    STATE.player.inExploration = true;
    this.renderMovement();
  },

  renderMovement() {
    const app = document.getElementById("app");
    app.innerHTML = `
      <div class="panel">
        <h2 style="color:var(--accent);">探索中 · ${CONFIG.MAPS.find(m=>m.id===STATE.player.currentMap).name}</h2>

const SEARCH = {
  gridW: 10,
  gridH: 20,
  currentItems: [],
  isRevealing: false,

  generateRandomItem(mapId) {
    const mapItems = CONFIG.ITEMS.filter(it => it.mapId === mapId);
    if (mapItems.length === 0) return null;
    const luck = STATE.player.luck;
    const weights = { common: 50, good: 30, rare: 15, epic: 4, legend: 1 };
    let w = { ...weights };
    if (luck > 5) { w.common -= 5; w.good += 2; w.rare += 2; w.epic += 1; }
    if (luck > 10) { w.common -= 5; w.good += 2; w.rare += 2; w.legend += 1; }
    const total = Object.values(w).reduce((s, v) => s + v, 0);
    let r = Math.random() * total;
    let quality = "common";
    for (const [qk, qw] of Object.entries(w)) { r -= qw; if (r <= 0) { quality = qk; break; } }
    const candidates = mapItems.filter(it => it.quality === quality);
    const base = candidates.length > 0 ? candidates[Math.floor(Math.random() * candidates.length)] : mapItems[Math.floor(Math.random() * mapItems.length)];
    return { ...base, uid: Date.now() + Math.random() };

  generatePointItems(mapId) {
    const count = 5 + Math.floor(Math.random() * 6);

      if (item) items.push(item);

  placeItems(items) {
    const grid = Array(this.gridH).fill(null).map(() => Array(this.gridW).fill(null));
    const placed = [];
    for (const item of items) {
      let placedOk = false;
      for (let attempts = 0; attempts < 100; attempts++) {
        const x = Math.floor(Math.random() * (this.gridW - item.width + 1));
        const y = Math.floor(Math.random() * (this.gridH - item.height + 1));
        let free = true;
        for (let dy = 0; dy < item.height; dy++) {
          for (let dx = 0; dx < item.width; dx++) {
            if (grid[y + dy][x + dx] !== null) { free = false; break; }
          }
          if (!free) break;
        }
        if (free) {
          for (let dy = 0; dy < item.height; dy++) {
            for (let dx = 0; dx < item.width; dx++) { grid[y + dy][x + dx] = item.uid; }
          }
          placed.push({ ...item, gx: x, gy: y });
          placedOk = true;
          break;
        }
      }
    }
    return { grid, placed };
  },

      const items = this.placed.filter(p => p.quality === qk);

  smartFillBag() {
    const bag = STATE.player.storageBag;
    const sorted = [...this.placed].sort((a, b) => b.perCell - a.perCell);
    let currentItems = [...bag.items];
    let currentUsed = currentItems.reduce((s, it) => s + it.width * it.height, 0);

    for (const candidate of sorted) {
      const cSize = candidate.width * candidate.height;
      if (currentUsed + cSize <= bag.capacity) {
        currentItems.push(candidate);
        currentUsed += cSize;
      } else {
        const currentSorted = currentItems.map((it, idx) => ({ ...it, idx })
        ).sort((a, b) => a.perCell - b.perCell);
        let freed = 0;
        const toRemove = [];
        for (const low of currentSorted) {
          if (freed >= cSize) break;
          freed += low.width * low.height;
          toRemove.push(low.idx);
        }
        if (freed >= cSize && candidate.perCell > currentSorted[0].perCell) {
          toRemove.sort((a, b) => b - a);
          toRemove.forEach(idx => currentItems.splice(idx, 1));
          currentItems.push(candidate);
          currentUsed = currentItems.reduce((s, it) => s + it.width * it.height, 0);
        }
      }
    }
    bag.items = currentItems;
    const el = document.getElementById("search-bag");
    if (el) el.innerHTML = `储物袋：${currentUsed}/${bag.capacity} 格`;
  }

    app.innerHTML = html;
  },

  sellAll() {
    const items = STATE.player.storageRing.items;
    let total = 0;
    items.forEach(it => total += it.baseValue);
    STATE.player.spiritStones += total;
    STATE.player.storageRing.items = [];
    alert(`出售成功，获得 ${total.toLocaleString()} 灵石`);
    this.render();
  }
};
```
